float_pattern = re.compile(r'[-+]?\d*\.\d+|\d+')

def extract_mean_floats(s):
    """Return a float from s. If s contains multiple numbers, return their mean."""
    if s is None:
        return np.nan
    if isinstance(s, (int, float, np.number)):
        return float(s)
    s_str = str(s).strip()
    # try direct conversion
    try:
        return float(s_str)
    except:
        pass
    # extract numeric substrings
    matches = float_pattern.findall(s_str)
    if not matches:
        return np.nan
    vals = [float(m) for m in matches]
    return float(np.mean(vals))

# ----------------- 3. Parse datetime safely -----------------
def safe_parse_datetime(row):
    sdate = str(row.get('Date','')).strip()
    stime = str(row.get('Time','')).strip()
    for fmt in ("%d-%m-%Y %H:%M:%S", "%Y-%m-%d %H:%M:%S", "%d/%m/%Y %H:%M:%S"):
        try:
            return datetime.strptime(f"{sdate} {stime}", fmt)
        except:
            pass
    try:
        return pd.to_datetime(sdate + " " + stime, dayfirst=True, errors='coerce')
    except:
        return pd.NaT

# ----------------- 6. Define Daily Time Slots and Rates & Calculate Per-Record TOU Cost -----------------

# Define the new daily time slots and their rates
daily_tariff_map = {
    (22, 6): {
      'slot': 'Night_Off_Peak',
      'rate': 3.75
      }, # 10 PM - 6 AM
    (6, 12): {
      'slot': 'Morning_Peak',
      'rate': 5.50
      },   # 6 AM - 12 PM
    (12, 18): {
      'slot': 'Forenoon_Normal',
      'rate': 4.75
      }, # 12 PM - 6 PM
    (18, 22): {
      'slot': 'Evening_Peak',
      'rate': 6.50
      }    # 6 PM - 10 PM
}

def get_daily_tariff(hour):
    for (start_hour, end_hour), tariff_info in daily_tariff_map.items():
        if start_hour < end_hour:
            if start_hour <= hour < end_hour:
                return tariff_info
        else: # Handles overnight slots like 22-6
            if hour >= start_hour or hour < end_hour:
                return tariff_info
    return {'slot': 'Unknown', 'rate': 0.0} # Fallback

# Apply the function to create new Tariff_Time_Slot and Tariff_Rate columns

df.loc[:, 'Tariff_Rate_Daily'] = df['Time_Hour'].apply(lambda x: get_daily_tariff(x)['rate'])

# Function for computing slab charge (re-used from weekly aggregation)
def compute_slab_charge(total_units, slabs):
    remaining = total_units
    prev_limit = 0.0
    charge = 0.0
    for limit, rate in slabs:
        if limit is None:
            charge += remaining * rate
            remaining = 0.0
            break
        slab_width = limit - prev_limit
        used = min(remaining, slab_width)
        charge += used * rate
        remaining -= used
        prev_limit = limit
        if remaining <= 0:
            break
    if remaining > 0:
        last_rate = slabs[-1][1]
        charge += remaining * last_rate
    return float(charge)

# Function for computing final slab bill (re-used from weekly aggregation)
def compute_final_slab_bill(total_units, slabs, fixed_charge=0.0, duty_pct=0.0, surcharge_pct=0.0, min_bill=None):
    energy_charge = compute_slab_charge(total_units, slabs)
    subtotal = energy_charge + fixed_charge
    duty = subtotal * (duty_pct / 100.0)
    surcharge = subtotal * (surcharge_pct / 100.0)
    total = subtotal + duty + surcharge
    if min_bill is not None:
        total = max(total, min_bill)
    return {
        'total_kwh': float(total_units),
        'energy_charge': energy_charge,
        'fixed_charge': fixed_charge,
        'duty': duty,
        'surcharge': surcharge,
        'final_bill': float(total)
    }

# Preprocessor & XGBoost helpers (re-defining for clarity in this block, but already present in kernel)
def build_preprocessor(num_features, cat_features):
    transformers = []
    if len(num_features) > 0:
        transformers.append(('num', StandardScaler(), num_features))
    if len(cat_features) > 0:
        transformers.append(('cat', OneHotEncoder(handle_unknown='ignore', sparse_output=False), cat_features))
    pre = ColumnTransformer(transformers=transformers, remainder='drop')
    return pre

def train_xgb(Xtr, ytr, Xte, params):
    model = xgb.XGBRegressor(**params, objective='reg:squarederror')
    model.fit(Xtr, ytr)
    preds = model.predict(Xte)
    return model, preds

def mape_pct(y_true, y_pred):
    return mean_absolute_percentage_error(y_true, y_pred) * 100.0
